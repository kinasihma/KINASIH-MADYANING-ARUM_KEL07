import java.util.*;

public class Main {
    public static void main(String[] args) {
      int x =10;
      System.out.println("Inc x = "+(++x));
      System.out.println("Dec x = "+(--x));

      // x += 10 artinya x baru = x + 10
      x += 10;
      System.out.println("x skrg = "+x);
      x -= 10;
      System.out.println("x skrg = "+x);
      x *= 5;
      System.out.println("x kali 5 = "+x);
      x /= 5;
      System.out.println("x bagi 5 = "+x);
      
      // operator : > < == != >=
      int a = 10;
      int b = 20;
      System.out.println("apakah a > b: "+(a>b));
      System.out.println("apakah a < b: "+(a<b));
      System.out.println("apakah a = b: "+(a==b));

      // operator logic: && (and) || (or) ! (not)
      boolean hasil = (a > b) && (a < b);
      System.out.println("Hasil dengan and = "+hasil);
      hasil = (a > b) || (a < b);
      System.out.println("Hasil dengan or = "+hasil);

      // PERCABANGAN
      //1. Kondisional / SELEKSI
      //2. Perulangan / LOOP

      System.out.println("a = "+a);
      int sisa = a % 2;
      if (sisa == 0) System.out.println("a genap");
      else System.out.println("a ganjil");

      // aplikasi login sederhana
      String username = "Iwan";
      String userinput = "Iwan";
      if (username == userinput) System.out.println("Login berhasil");
      else System.out.println("Login gagal");

      a = 5;
      b = 10;
      int c = 2;
      int d = b*b-4*a*c;
      System.out.println("Diskriminan = "+d);
      if (d >= 0) System.out.println("Akar Real positif");
      else System.out.println("Akar imajiner");

      // nested if
      int golongan = 6;
      if (golongan == 1) System.out.println("Gaji anda 1 juta");
      else if (golongan == 2) System.out.println("Gaji anda 2 juta");
      else if (golongan == 3) System.out.println("Gaji anda 3 juta");
      else if (golongan == 4) System.out.println("Gaji anda 4 juta");
      else if (golongan == 5) System.out.println("Gaji anda 5 juta");
      else System.out.println("Input golongan salah");

      for (int i = 1; i < 4; i++) {
      System.out.println(i);
      }
      System.out.println("Say cheese!");

      int countdown = 3;

      while (countdown > 0) {
      System.out.println(countdown);
      countdown--;
      }
      
      System.out.println("Happy New Year!!");
    }
}
